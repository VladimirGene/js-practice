// Готовая функция для проверки. Временно замени fetch(url) на mockFetchPosts(window.postsScenario).
// Перед нажатием на кнопку задай в консоли window.postsScenario = "empty", чтобы проверить пустой список.
// Используй "success" для карточек, "http-error" для HTTP 500 и "network-error" для ошибки соединения.
// Чтобы проверить повторную попытку, сначала выбери "http-error". После ошибки переключись на "success"
// без обновления страницы и нажми "Попробовать снова". Должны появиться карточки.
function mockFetchPosts(scenario = "success") {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (scenario === "network-error") {
        reject(new TypeError("Ошибка соединения"));
        return;
      }

      if (scenario === "http-error") {
        resolve(new Response(null, {status: 500}));
        return;
      }

      const posts = scenario === "empty" ? [] : [
        {id: 1, title: "Первый пост", body: "Текст первого поста."},
        {id: 2, title: "Второй пост", body: "Текст второго поста."}
      ];

      resolve(new Response(JSON.stringify(posts), {
        headers: {"Content-Type": "application/json"}
      }));
    }, 1000);
  });
